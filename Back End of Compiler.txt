#include <stdio.h>

int main()
{
    char op;
    int a, b;

    printf("Enter first number: ");
    scanf("%d", &a);

    printf("Enter operator (+,-,*,/): ");
    scanf(" %c", &op);

    printf("Enter second number: ");
    scanf("%d", &b);

    printf("\nIntermediate Code: t1 = %d %c %d\n", a, op, b);

    printf("\nGenerated Assembly Code:\n");

    printf("MOV R1, %d\n", a);
    printf("MOV R2, %d\n", b);

    if (op == '+')
        printf("ADD R1, R2\n");
    else if (op == '-')
        printf("SUB R1, R2\n");
    else if (op == '*')
        printf("MUL R1, R2\n");
    else if (op == '/')
        printf("DIV R1, R2\n");

    printf("MOV RESULT, R1\n");

    return 0;
}
